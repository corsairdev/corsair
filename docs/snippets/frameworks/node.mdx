Raw `node:http` has no dedicated adapter, so most Node apps use [Express](/frameworks/express) or [Hono](/frameworks/hono) (both run on Node). If you need bare `http`, bridge the Web-standard handler:

```ts server.ts
import { createServer } from 'node:http';
import { managementHandler } from 'corsair';
import { corsair } from './corsair';

const handler = managementHandler(corsair, { basePath: '/api/corsair' });

createServer(async (req, res) => {
    const url = `http://${req.headers.host}${req.url}`;
    const chunks: Buffer[] = [];
    for await (const chunk of req) chunks.push(chunk as Buffer);
    const body = chunks.length ? Buffer.concat(chunks) : undefined;

    const response = await handler(
        new Request(url, {
            method: req.method,
            headers: req.headers as HeadersInit,
            body,
        }),
    );

    res.writeHead(response.status, Object.fromEntries(response.headers));
    res.end(Buffer.from(await response.arrayBuffer()));
}).listen(3000);
```

<Note>Node 18+ ships global `Request`/`Response`. On older Node, install `undici` and import them from there.</Note>
