Onboard a user to an integration by creating a connect link and redirecting. Corsair Hub hosts the consent screen, runs the OAuth handshake, and delivers the encrypted tokens back to your route.

<Tabs items={["React", "Vanilla JS"]}>
  <Tab value="React">

```tsx title="connect-github.tsx"
"use client";
import { useConnectionStatus, useCreateConnectLink } from "./corsair-client";

export function ConnectGitHub({ tenantId }: { tenantId: string }) {
    const { mutate, loading } = useCreateConnectLink();
    const { data: status } = useConnectionStatus({ tenantId });

    if (status?.github === "connected") return <span>GitHub connected ✓</span>;

    return (
        <button
            disabled={loading}
            onClick={async () => {
                const link = await mutate({ plugin: "github", tenantId });
                if (link) window.location.href = link.connectUrl;
            }}
        >
            Connect GitHub
        </button>
    );
}
```

  </Tab>
  <Tab value="Vanilla JS">

```ts title="connect.ts"
// create a connect link and send the user to it
const { connectUrl } = await client.connect.createLink({
    plugin: "github",
    tenantId: "acme",
});
window.location.href = connectUrl;

// after they return, the status flips to "connected"
const status = await client.connectionStatus.get({ tenantId: "acme" });
// { github: 'connected', slack: 'not_connected', ... }
```
